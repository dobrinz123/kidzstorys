```typescript
import { convexTest } from "convex-test";
import { describe, test, expect, vi, beforeEach, afterEach } from "vitest";
import { internal } from "./_generated/api";
import schema from "./schema";

// Helper: minimal valid category insert
async function insertCategory(ctx: any) {
  return await ctx.db.insert("categories", {
    slug: "test-cat",
    order: 1,
    name_en: "Test", name_ro: "Test", name_it: "Test",
    description_en: "", description_ro: "", description_it: "",
    iconKey: "test",
    isActive: true,
    createdAt: Date.now(),
    updatedAt: Date.now(),
  });
}

// Helper: minimal valid book insert
async function insertBook(ctx: any, categoryId: any, pageTextEn = "Hello world.") {
  return await ctx.db.insert("books", {
    bookKey: "test-book-" + Date.now(),
    categoryId,
    ageMin: 3, ageMax: 6, durationMin: 5,
    cover: { title_en: "Test", title_ro: "Test", title_it: "Test" },
    pages: [{
      pageNumber: 1,
      text_en: pageTextEn,
      text_ro: "Salut lume.",
      text_it: "Ciao mondo.",
    }],
    tags_en: [], tags_ro: [], tags_it: [],
    parentQuestions_en: [], parentQuestions_ro: [], parentQuestions_it: [],
    status: "draft",
    version: 1,
    readCount: 0,
    favoriteCount: 0,
    createdAt: Date.now(),
    updatedAt: Date.now(),
  });
}

// Helper: minimal valid audioJob insert
async function insertJob(ctx: any, bookId: any, locale: "en" | "ro") {
  return await ctx.db.insert("audioJobs", {
    bookId,
    pageIndex: 0,
    locale,
    status: "pending" as const,
    retryCount: 0,
    createdAt: Date.now(),
  });
}

// Mock fetch that returns a minimal MP3 blob (0xff 0xfb = MP3 frame sync)
function mockFetchSuccess() {
  vi.stubGlobal("fetch", vi.fn(async (url: string) => {
    if (url.includes("api.elevenlabs.io")) {
      return new Response(new Uint8Array([0xff, 0xfb, 0x90, 0x00]), {
        status: 200,
        headers: { "Content-Type": "application/octet-stream" },
      });
    }
    throw new Error(`Unexpected fetch to: ${url}`);
  }));
}

// Mock fetch that returns a 422 error
function mockFetch422() {
  vi.stubGlobal("fetch", vi.fn(async (url: string) => {
    if (url.includes("api.elevenlabs.io")) {
      return new Response(JSON.stringify({ detail: "invalid voice" }), {
        status: 422,
        headers: { "Content-Type": "application/json" },
      });
    }
    throw new Error(`Unexpected fetch to: ${url}`);
  }));
}

beforeEach(() => {
  // Stub env vars — convex-test does not inject real Convex dashboard env vars
  process.env.ELEVENLABS_API_KEY = "test-api-key";
  process.env.ELEVENLABS_EN_VOICE_ID = "voice-en-test";
  process.env.ELEVENLABS_RO_VOICE_ID = "voice-ro-test";
});

afterEach(() => {
  vi.unstubAllGlobals();
  delete process.env.ELEVENLABS_API_KEY;
  delete process.env.ELEVENLABS_EN_VOICE_ID;
  delete process.env.ELEVENLABS_RO_VOICE_ID;
});

describe("generatePageAudio", () => {
  test("happy path EN: job and page marked done with URL", async () => {
    mockFetchSuccess();
    const t = convexTest(schema, import.meta.glob("./**/*.ts"));

    const { bookId, jobId } = await t.run(async (ctx) => {
      const categoryId = await insertCategory(ctx);
      const bookId = await insertBook(ctx, categoryId, "Hello world.");
      const jobId = await insertJob(ctx, bookId, "en");
      return { bookId, jobId };
    });

    await t.action(internal.audio.generatePageAudio, {
      jobId, bookId, pageIndex: 0, locale: "en",
    });

    const job = await t.run(async (ctx) => ctx.db.get(jobId));
    expect(job?.status).toBe("done");
    expect(job?.audioUrl).toBeTruthy();
    expect(job?.audioStorageId).toBeTruthy();
    expect(job?.completedAt).toBeGreaterThan(0);

    const book = await t.run(async (ctx) => ctx.db.get(bookId));
    expect(book?.pages[0].audioStatus_en).toBe("done");
    expect(book?.pages[0].audioUrl_en).toBeTruthy();
    expect(book?.pages[0].audioStorageId_en).toBeTruthy();
  });

  test("happy path RO: job and page marked done with URL", async () => {
    mockFetchSuccess();
    const t = convexTest(schema, import.meta.glob("./**/*.ts"));

    const { bookId, jobId } = await t.run(async (ctx) => {
      const categoryId = await insertCategory(ctx);
      const bookId = await insertBook(ctx, categoryId);
      const jobId = await insertJob(ctx, bookId, "ro");
      return { bookId, jobId };
    });

    await t.action(internal.audio.generatePageAudio, {
      jobId, bookId, pageIndex: 0, locale: "ro",
    });

    const job = await t.run(async (ctx) => ctx.db.get(jobId));
    expect(job?.status).toBe("done");
    expect(job?.audioUrl).toBeTruthy();

    const book = await t.run(async (ctx) => ctx.db.get(bookId));
    expect(book?.pages[0].audioStatus_ro).toBe("done");
    expect(book?.pages[0].audioUrl_ro).toBeTruthy();

    // Confirm RO voice ID was used in the fetch URL
    const fetchMock = vi.mocked(fetch);
    const calledUrl = fetchMock.mock.calls[0][0] as string;
    expect(calledUrl).toContain("voice-ro-test");
  });

  test("ElevenLabs 4xx: job marked failed with error message", async () => {
    mockFetch422();
    const t = convexTest(schema, import.meta.glob("./**/*.ts"));

    const { bookId, jobId } = await t.run(async (ctx) => {
      const categoryId = await insertCategory(ctx);
      const bookId = await insertBook(ctx, categoryId);
      const jobId = await insertJob(ctx, bookId, "en");
      return { bookId, jobId };
    });

    await expect(
      t.action(internal.audio.generatePageAudio, {
        jobId, bookId, pageIndex: 0, locale: "en",
      })
    ).rejects.toThrow();

    const job = await t.run(async (ctx) => ctx.db.get(jobId));
    expect(job?.status).toBe("failed");
    expect(job?.errorMessage).toMatch(/422/);

    const book = await t.run(async (ctx) => ctx.db.get(bookId));
    expect(book?.pages[0].audioStatus_en).toBe("failed");
  });

  test("empty page text: job marked failed, fetch never called", async () => {
    const fetchMock = vi.fn();
    vi.stubGlobal("fetch", fetchMock);
    const t = convexTest(schema, import.meta.glob("./**/*.ts"));

    const { bookId, jobId } = await t.run(async (ctx) => {
      const categoryId = await insertCategory(ctx);
      // Insert book with empty EN text on page 0
      const bookId = await insertBook(ctx, categoryId, "");
      const jobId = await insertJob(ctx, bookId, "en");
      return { bookId, jobId };
    });

    await expect(
      t.action(internal.audio.generatePageAudio, {
        jobId, bookId, pageIndex: 0, locale: "en",
      })
    ).rejects.toThrow();

    // fetch must NOT have been called — action should fail before reaching ElevenLabs
    expect(fetchMock).not.toHaveBeenCalled();

    const job = await t.run(async (ctx) => ctx.db.get(jobId));
    expect(job?.status).toBe("failed");
    expect(job?.errorMessage).toBeTruthy();
  });
});
```

If any type errors arise with `ctx.db.get` return type assertions, use optional chaining (already shown above with `job?.status`).
